#include <tunables/global>

# Supervisor renames this profile to the installed add-on's full slug when it loads it.
profile whatsapp_gateway flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities s6-overlay uses while preparing /run and dropping into services
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,

  network,
  unix,
  signal,

  # s6-overlay: /init and its helpers are shell/execline scripts, so they need
  # read as well as exec permission.
  /init rix,
  /bin/** rix,
  /sbin/** rix,
  /usr/bin/** rix,
  /usr/sbin/** rix,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** rix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwkix,
  /var/run/{,**} rwk,
  /dev/tty rw,

  # Runtime libraries, configuration and kernel information
  /lib/** rm,
  /usr/lib/** rm,
  /usr/local/lib/** rm,
  /usr/share/** r,
  /etc/** r,
  /dev/null rw,
  /dev/zero rw,
  /dev/random r,
  /dev/urandom r,
  @{PROC}/** r,
  /sys/** r,
  /tmp/{,**} rwk,

  # Application code (read only; native modules are mapped)
  /app/{,**} rm,

  # Persistent state: options, Baileys auth, groups cache
  /data/{,**} rwk,
}
